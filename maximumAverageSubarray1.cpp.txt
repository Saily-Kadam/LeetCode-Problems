class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        int sz = nums.size();
        int ans = 0;
        int avg = 0;
        for(int i=0; i<k; i++)
            ans += nums[i];
            avg = ans;
        for(int i=1; i<=sz-k; i++){
            ans -= nums[i-1];
            ans += nums[i+k - 1];
            avg = max(avg,ans);
        }  
        return avg/double(k);  
    }
};
